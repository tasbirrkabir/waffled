    68→function CheckoutSteps({ step, done = false }: { step: Step; done?: boolean }) {
    69→  const steps = [

    71→    { n: 2, label: "Checkout" },

    75→  return (
    76→    <ol className="flex flex-wrap items-center gap-x-2 gap-y-2 text-sm font-medium" aria-label="Checkout progress">
    77→      {steps.map((s, i) => {
    78→        const isDone = current > s.n;
    79→        const isCurrent = current === s.n;
    80→        return (
    81→          <li key={s.n} className="flex items-center gap-2">
    82→            <span
    83→              className={cn(
    84→                "flex h-7 w-7 items-center justify-center rounded-full border-2 font-display text-xs font-bold",
    85→                isDone
    86→                  ? "border-lime bg-lime text-ink"
    87→                  : isCurrent
    88→                    ? "border-ink bg-ink text-cream"
    89→                    : "border-ink/25 bg-white text-muted-warm"
    90→              )}
    91→              aria-current={isCurrent ? "step" : undefined}
    92→            >
    93→              {isDone ? <Check className="h-3.5 w-3.5" strokeWidth={3.5} aria-hidden /> : s.n}
    94→            </span>
    95→            <span className={cn(isCurrent ? "text-ink" : "text-muted-warm")}>{s.label}</span>
    96→            {i < steps.length - 1 && (
    97→              <span aria-hidden className="ml-1 text-ink/25">
    98→                →
    99→              </span>
   100→            )}
   101→          </li>
   102→        );

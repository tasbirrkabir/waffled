    57→/* ------------------------------- Step header ------------------------------ */
    58→
    59→function CheckoutSteps({ step, done = false }: { step: Step; done?: boolean }) {
    60→  const steps = [
    61→    { n: 1, label: "Your items" },
    62→    { n: 2, label: "Checkout" },
    63→    { n: 3, label: "Confirmed" },
    64→  ];
    65→  const current = done ? 3 : step;
    66→  return (
    67→    <ol className="flex flex-wrap items-center gap-x-2 gap-y-2 text-sm font-medium" aria-label="Checkout progress">
    68→      {steps.map((s, i) => {
    69→        const isDone = current > s.n;
    70→        const isCurrent = current === s.n;
    71→        return (
    72→          <li key={s.n} className="flex items-center gap-2">
    73→            <span
    74→              className={cn(
    75→                "flex h-7 w-7 items-center justify-center rounded-full border-2 font-display text-xs font-bold",
    76→                isDone
    77→                  ? "border-lime bg-lime text-ink"
    78→                  : isCurrent
    79→                    ? "border-ink bg-ink text-cream"
    80→                    : "border-ink/25 bg-white text-muted-warm"
    81→              )}
    82→              aria-current={isCurrent ? "step" : undefined}
    83→            >
    84→              {isDone ? <Check className="h-3.5 w-3.5" strokeWidth={3.5} aria-hidden /> : s.n}
    85→            </span>
    86→            <span className={cn(isCurrent ? "text-ink" : "text-muted-warm")}>{s.label}</span>
    87→            {i < steps.length - 1 && (
    88→              <span aria-hidden className="ml-1 text-ink/25">
    89→                →
    90→              </span>
    91→            )}
    92→          </li>
    93→        );
    94→      })}
    95→    </ol>
    96→  );
    97→}

   157→
   158→export function OrderPageContent() {
   159→  const lines = useCart((s) => s.lines);
   160→  const increment = useCart((s) => s.increment);
   161→  const decrement = useCart((s) => s.decrement);
   162→  const remove = useCart((s) => s.remove);
   163→  const clear = useCart((s) => s.clear);
   164→
   165→  const [step, setStep] = React.useState<Step>(1);
   166→  const [activeCat, setActiveCat] = React.useState(menu[0].id);
   167→  const [name, setName] = React.useState("");
   168→  const [phone, setPhone] = React.useState("");
   169→  const [pickupTime, setPickupTime] = React.useState<string>(pickupOptions[0]);
   170→  const [notes, setNotes] = React.useState("");
   171→  const [submitting, setSubmitting] = React.useState(false);
   172→  const [success, setSuccess] = React.useState<OrderSuccess | null>(null);
   173→  const [touched, setTouched] = React.useState<{ name?: boolean; phone?: boolean }>({});
   174→
   175→  const subtotal = cartSubtotal(lines);
   176→  const itemCount = lines.reduce((n, l) => n + l.qty, 0);
   177→  const nameValid = name.trim().length >= 2;
   178→  const phoneValid = validatePhone(phone);
   179→  const canSubmit = lines.length > 0 && nameValid && phoneValid && !submitting;
   180→
   181→  const category = menu.find((c) => c.id === activeCat) ?? menu[0];
   182→
   183→  function goToStep(next: Step) {
   184→    setStep(next);
   185→    window.scrollTo({ top: 0, behavior: "smooth" });
   186→  }
   187→
   188→  async function placeOrder() {
   189→    setTouched({ name: true, phone: true });
   190→    if (!canSubmit) {
   191→      if (lines.length === 0) toast.error("Add at least one item to your order first.");
   192→      else if (!nameValid) toast.error("Please tell us your name.");
   193→      else if (!phoneValid) toast.error("Please enter a valid Bangladeshi phone number.");
   194→      return;
   195→    }
   196→    setSubmitting(true);
   197→    try {
   198→      const res = await fetch("/api/orders", {
   199→        method: "POST",
   200→        headers: { "Content-Type": "application/json" },
   201→        body: JSON.stringify({
   202→          customerName: name.trim(),
   203→          phone: phone.trim(),
   204→          pickupTime,
   205→          notes: notes.trim() || undefined,
   206→          items: lines.map((l) => ({ id: l.id, qty: l.qty })),
   207→        }),
   208→      });
   209→      const data = await res.json();
   210→      if (!res.ok || !data.success) {
   211→        throw new Error(data.error || "Could not place the order.");
   212→      }
   213→      setSuccess({
   214→        orderNumber: data.orderNumber,
   215→        total: data.total,
   216→        whatsappUrl: data.whatsappUrl,
   217→        lines,
   218→        name: name.trim(),
   219→      });
   220→      clear();
   221→      window.scrollTo({ top: 0, behavior: "smooth" });
   222→    } catch (err) {
   223→      toast.error(err instanceof Error ? err.message : "Could not place the order. Please try again.");
   224→    } finally {
   225→      setSubmitting(false);
   226→    }
   227→  }
   228→
   229→  /* ------------------------------ SUCCESS VIEW ----------------------------- */
   230→  if (success) {
   231→    return (
   232→      <div className="mx-auto max-w-2xl px-4 pb-24 pt-24 sm:px-6 md:pt-28">
   233→        <div className="mb-6">
   234→          <CheckoutSteps step={2} done />
   235→        </div>
   236→        <motion.div
   237→          initial={{ opacity: 0, y: 20 }}
   238→          animate={{ opacity: 1, y: 0 }}
   239→          className="rounded-3xl border-[3px] border-ink bg-white p-6 text-center shadow-[8px_8px_0_0_var(--color-ink)] sm:p-10"
   240→        >
   241→          <span className="mx-auto flex h-16 w-16 items-center justify-center rounded-full border-[3px] border-ink bg-lime shadow-[3px_3px_0_0_var(--color-ink)]">
   242→            <Check className="h-8 w-8 text-ink" strokeWidth={3} aria-hidden />
   243→          </span>
   244→          <h1 className="mt-5 font-display text-3xl font-bold sm:text-4xl">
   245→            Order placed!
   246→          </h1>
   247→          <p className="mt-2 text-muted-warm">
   248→            Thanks {success.name.split(" ")[0]}. Your order number is{" "}
   249→            <span className="font-display font-bold text-pink">{success.orderNumber}</span>,
   250→            total {formatTk(success.total)}.
   251→          </p>
   252→
   253→          {/* Item recap */}
   254→          <ul className="mt-6 space-y-2 rounded-2xl border-2 border-dashed border-ink/25 bg-cream p-4 text-left">
   255→            {success.lines.map((l) => (
   256→              <li key={l.id} className="flex items-center justify-between gap-3 text-sm">
   257→                <span className="font-semibold text-ink">
   258→                  {l.qty} × {l.name}
   259→                </span>
   260→                <span className="font-bold tabular-nums text-ink/80">{formatTk(l.price * l.qty)}</span>
   261→              </li>
   262→            ))}
   263→            <li className="flex items-center justify-between border-t-2 border-dashed border-ink/20 pt-2 text-sm">
   264→              <span className="font-bold text-ink">Total (pay at pickup)</span>
   265→              <span className="font-display font-bold tabular-nums text-pink">
   266→                {formatTk(success.total)}
   267→              </span>
   268→            </li>
   269→          </ul>
   270→
   271→          <div className="mt-5 rounded-2xl border-2 border-dashed border-ink/30 bg-cream p-4 text-left">
   272→            <p className="flex items-start gap-2 text-sm font-semibold text-ink">
   273→              <Clock className="mt-0.5 h-4 w-4 shrink-0 text-orange" aria-hidden />
   274→              One last step: tap the WhatsApp button below so the shop receives your
   275→              order instantly. You can also call us to confirm.
   276→            </p>
   277→          </div>
   278→
   279→          <div className="mt-6 flex flex-col justify-center gap-3 sm:flex-row">
   280→            <Button
   281→              asChild
   282→              className="min-h-12 rounded-full bg-[#25D366] px-7 text-base font-bold text-white shadow-[4px_4px_0_0_var(--color-ink)] transition-transform hover:translate-x-1 hover:translate-y-1 hover:shadow-none"
   283→            >
   284→              <a href={success.whatsappUrl} target="_blank" rel="noopener noreferrer">
   285→                Send order on WhatsApp
   286→              </a>
   287→            </Button>
   288→            <Button
   289→              asChild
   290→              variant="outline"
   291→              className="min-h-12 rounded-full border-2 border-ink bg-white px-7 text-base font-semibold text-ink"
   292→            >
   293→              <a href={`tel:${siteConfig.phoneE164}`} onClick={() => trackEvent("phone_click")}>
   294→                <Phone className="mr-1 h-4 w-4" aria-hidden />
   295→                Call the shop
   296→              </a>
   297→            </Button>
   298→          </div>
   299→
   300→          <button
   301→            type="button"
   302→            onClick={() => {
   303→              setSuccess(null);
   304→              setStep(1);
   305→              setName("");
   306→              setPhone("");
   307→              setNotes("");
   308→              setTouched({});
   309→            }}
   310→            className="mt-6 text-sm font-semibold text-pink underline-offset-4 hover:underline"
   311→          >
   312→            Start a new order
   313→          </button>
   314→        </motion.div>
   315→      </div>
   316→    );
   317→  }
   318→
   319→  /* ------------------------------- STEP 2: CHECKOUT ------------------------ */
   320→  if (step === 2) {
   321→    return (
   322→      <div className="mx-auto max-w-6xl px-4 pb-32 pt-24 sm:px-6 md:pb-24 md:pt-28">
   323→        {/* Compact header */}
   324→        <div className="flex flex-wrap items-center justify-between gap-3">
   325→          <div>
   326→            <button
   327→              type="button"
   328→              onClick={() => goToStep(1)}
   329→              className="mb-2 inline-flex min-h-11 items-center gap-1.5 rounded-full px-3 text-sm font-semibold text-muted-warm transition-colors hover:bg-white hover:text-ink"
   330→            >
   331→              <ArrowLeft className="h-4 w-4" aria-hidden />
   332→              Back to items
   333→            </button>
   334→            <h1 className="font-display text-3xl font-bold tracking-tight sm:text-4xl">
   335→              Checkout
   336→            </h1>
   337→          </div>
   338→          <CheckoutSteps step={2} />
   339→        </div>
   340→
   341→        <div className="mt-6 grid gap-6 lg:grid-cols-[1fr_400px] lg:gap-8">
   342→          {/* Details form */}
   343→          <div className="space-y-6">
   344→            <section
   345→              aria-labelledby="details-heading"
   346→              className="rounded-3xl border-2 border-ink/15 bg-white p-5 sm:p-6"
   347→            >
   348→              <h2 id="details-heading" className="flex items-center gap-2 font-display text-lg font-bold">
   349→                <span className="flex h-7 w-7 items-center justify-center rounded-full bg-pink font-display text-xs font-bold text-white">
   350→                  1
   351→                </span>
   352→                Your details
   353→              </h2>
   354→              <div className="mt-4 grid gap-4 sm:grid-cols-2">
   355→                <div>
   356→                  <Label htmlFor="order-name" className="text-sm font-semibold">
   357→                    Name
   358→                  </Label>
   359→                  <Input
   360→                    id="order-name"
   361→                    value={name}
   362→                    onChange={(e) => setName(e.target.value)}
   363→                    onBlur={() => setTouched((t) => ({ ...t, name: true }))}
   364→                    placeholder="e.g. Rahim Ahmed"
   365→                    autoComplete="name"
   366→                    maxLength={60}
   367→                    required
   368→                    aria-invalid={touched.name && !nameValid}
   369→                    className="mt-1.5 h-12 border-2 border-ink/25 bg-cream focus-visible:ring-pink"
   370→                  />

   372→                    <p className="mt-1 text-xs font-medium text-red-600">

   376→                </div>
   377→                <div>

   379→                    Phone number
   380→                  </Label>
   381→                  <Input
   382→                    id="order-phone"
   383→                    type="tel"
   384→                    inputMode="tel"
   385→                    value={phone}
   386→                    onChange={(e) => setPhone(e.target.value)}
   387→                    onBlur={() => setTouched((t) => ({ ...t, phone: true }))}
   388→                    placeholder="01XXXXXXXXX"
   389→                    autoComplete="tel"
   390→                    maxLength={16}
   391→                    required
   392→                    aria-invalid={touched.phone && !phoneValid}
   393→                    className="mt-1.5 h-12 border-2 border-ink/25 bg-cream focus-visible:ring-pink"
   394→                  />
   395→                  {touched.phone && !phoneValid && (
   396→                    <p className="mt-1 text-xs font-medium text-red-600">
   397→                      Enter a valid BD number, e.g. 01723399638.
   398→                    </p>
   399→                  )}
   400→                </div>
   401→              </div>
   402→            </section>
   403→
   404→            <section
   405→              aria-labelledby="pickup-heading"
   406→              className="rounded-3xl border-2 border-ink/15 bg-white p-5 sm:p-6"
   407→            >
   408→              <h2 id="pickup-heading" className="flex items-center gap-2 font-display text-lg font-bold">
   409→                <span className="flex h-7 w-7 items-center justify-center rounded-full bg-cyan font-display text-xs font-bold text-ink">
   410→                  2
   411→                </span>
   412→                Pickup
   413→              </h2>
   414→              <div className="mt-4 grid gap-4">
   415→                <div>
   416→                  <Label htmlFor="order-pickup" className="text-sm font-semibold">
   417→                    Pickup time

   434→                    Notes <span className="font-normal text-muted-warm">(optional)</span>

   436→                  <Textarea
   437→                    id="order-notes"
   438→                    value={notes}
   439→                    onChange={(e) => setNotes(e.target.value)}

   443→                  />

   446→            </section>
   447→
   448→            <section
   449→              aria-label="How pickup works"

   451→            >
   452→              <div className="flex items-start gap-3">
   453→                <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full border-2 border-ink bg-lime-soft">
   454→                  <Store className="h-5 w-5 text-ink" aria-hidden />
   455→                </span>
   456→                <div>
   457→                  <p className="text-sm font-bold">Pickup only</p>
   458→                  <p className="mt-0.5 text-xs leading-relaxed text-muted-warm">
   459→                    {siteConfig.address.street}, {siteConfig.address.locality}{" "}
   460→                    {siteConfig.address.postalCode}
   461→                  </p>
   462→                </div>
   463→              </div>
   464→              <div className="flex items-start gap-3">
   465→                <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full border-2 border-ink bg-orange-soft">
   466→                  <Wallet className="h-5 w-5 text-ink" aria-hidden />
   467→                </span>
   468→                <div>
   469→                  <p className="text-sm font-bold">Pay at the counter</p>

   472→                  </p>
   473→                </div>
   474→              </div>
   475→            </section>
   476→
   477→            {/* Order summary (mobile / tablet) */}
   478→            <section
   479→              aria-labelledby="summary-heading-mobile"
   480→              className="rounded-3xl border-[3px] border-ink bg-white p-5 shadow-[6px_6px_0_0_var(--color-ink)] lg:hidden"
   481→            >
   482→              <h2 id="summary-heading-mobile" className="flex items-center gap-2 font-display text-lg font-bold">
   483→                <ShoppingBag className="h-5 w-5 text-pink" aria-hidden />
   484→                Order summary
   485→                {itemCount > 0 && (
   486→                  <span className="rounded-full bg-pink px-2.5 py-0.5 text-xs font-bold text-white tabular-nums">
   487→                    {itemCount}
   488→                  </span>

   490→              </h2>
   491→              {lines.length === 0 ? (
   492→                <div className="mt-4 rounded-2xl border-2 border-dashed border-ink/25 p-6 text-center">
   493→                  <p className="text-sm font-medium text-muted-warm">
   494→                    Your order is empty.{" "}
   495→                    <button
   496→                      type="button"
   497→                      onClick={() => goToStep(1)}
   498→                      className="font-bold text-pink underline-offset-2 hover:underline"
   499→                    >
   500→                      Add some treats
   501→                    </button>
   502→                  </p>

   507→                    {lines.map((line) => (
   508→                      <CartLineRow
   509→                        key={line.id}
   510→                        line={line}
   511→                        onInc={increment}
   512→                        onDec={decrement}
   513→                        onRemove={remove}
   514→                        compact
   515→                      />
   516→                    ))}
   517→                  </ul>
   518→                  <div className="mt-4 flex items-center justify-between border-t-2 border-dashed border-ink/20 pt-3">
   519→                    <span className="text-sm font-semibold text-muted-warm">Total</span>
